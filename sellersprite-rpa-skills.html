安装配置

快速开始亚马逊数据分析

前置要求
# Kimi WebBridge daemon运行且扩展已连接
~/.kimi-webbridge/bin/kimi-webbridge status
# 期望：running: true, extension_connected: true

# Python 3.10+（无第三方依赖）
# 已登录sellersprite的浏览器
快速开始
# 1. 验证守护进程
~/.kimi-webbridge/bin/kimi-webbridge status

# 2. 选产品 — 美国站Top 100 ASIN
python .claude/skills/sellersprite-products/scripts/fetch_products.py \
 --market US --pages 5 --out data/us_top100.csv

# 3. 关键词选品 — 3站点flashlight搜索数据
python .claude/skills/sellersprite-keywords/scripts/fetch_keywords.py \
 --station US,JP,GB --keyword flashlight --pages 2 \
 --out data/keywords_by_station.csv

# 4. 生成对比报告
python .claude/skills/sellersprite-products/scripts/analyze.py \
 --products data/us_top100.csv --out-md reports/products.md
